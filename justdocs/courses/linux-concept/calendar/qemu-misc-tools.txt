#install software

####
# 生成一個空白的 8g 檔案準備給 FreeBSD 作業系統使用
# rem create empty 8G image #20g too big, cannot save at google drive
qemu-img.exe create -f raw FreeBSD.img 8g

####
# 安裝作業系統
# install into empty FreeBSD.img
qemu-system-x86_64.exe -m 2g -smp 2 -drive file=FreeBSD.img,format=raw -cdrom FreeBSD-bootOnly.iso -boot d
  
####
# 開機
# simulate FreeBSD minimize parameters #nor cdrom or boot 
qemu-system-x86_64.exe -m 2g -smp 2 -drive file=FreeBSD.img,format=raw

####
# at console add a new user
adduser

####
# 安全登入改埠號
# redirect port 22 to 2222
qemu-system-x86_64.exe -m 2g -smp 2 -drive file=FreeBSD.img,format=raw -nic user,hostfwd=tcp:127.0.0.1:2222-:22

####
# 由串接埠登入
# console to serial port 
qemu-system-x86_64.exe -m 2g -smp 2 -drive file=FreeBSD.img,format=raw -nic user,hostfwd=tcp:127.0.0.1:2222-:22 -nographic

####
#螢幕和串接埠二者皆可登入系統
#For maximum flexibility, console="comconsole,vidconsole" is the better setting.
echo 'console="comconsole,vidconsole"' > /boot/loader.conf
